    <script>
        lucide.createIcons();

        if (document.fullscreenEnabled || document.webkitFullscreenEnabled) {
            document.getElementById('btn-fullscreen').classList.remove('hidden');
        }

        const state = {
            isRunning: false,
            isFrozen: false,
            minFreq: 20,
            maxFreq: 22050,
            animationId: null,
            peakHz: 0,
            peakMag: 0,
            isHovering: false,
            hoverX: 0
        };

        const rtaCanvas = document.getElementById('rta-canvas');
        const rtaCtx = rtaCanvas.getContext('2d', { alpha: false }); 
        
        const waterfallCanvas = document.getElementById('waterfall-canvas');
        const waterfallCtx = waterfallCanvas.getContext('2d', { alpha: false });

        const btnStart = document.getElementById('btn-start');
        const startOverlay = document.getElementById('start-overlay');
        const btnFreeze = document.getElementById('btn-freeze');
        const iconFreeze = document.getElementById('icon-freeze');
        const freezeGlow = document.getElementById('freeze-glow');
        const btnFullscreen = document.getElementById('btn-fullscreen');
        const iconFullscreen = document.getElementById('icon-fullscreen');
        
        const valPeakFreq = document.getElementById('peak-freq-val');
        const valPeakMag = document.getElementById('peak-mag-val');
        const freqLabelsContainer = document.getElementById('freq-labels-container');

        let audioCtx = null;
        let analyser = null;
        let dataArray = null;      
        let peakArray = null;      

        // --- SISTEMA DE INTERACTIVIDAD TÁCTIL Y MOUSE ---
        function updateHoverPosition(e) {
            const rect = rtaCanvas.getBoundingClientRect();
            let clientX = e.clientX;
            if (e.touches && e.touches.length > 0) {
                clientX = e.touches[0].clientX;
            }
            
            if (clientX >= rect.left && clientX <= rect.right) {
                state.isHovering = true;
                state.hoverX = clientX - rect.left;
            } else {
                state.isHovering = false;
            }
        }

        rtaCanvas.addEventListener('mousemove', updateHoverPosition);
        rtaCanvas.addEventListener('mouseleave', () => state.isHovering = false);
        
        rtaCanvas.addEventListener('touchstart', (e) => { 
            e.preventDefault(); 
            updateHoverPosition(e); 
        }, { passive: false });
        
        rtaCanvas.addEventListener('touchmove', (e) => { 
            e.preventDefault(); 
            updateHoverPosition(e); 
        }, { passive: false });
        
        rtaCanvas.addEventListener('touchend', () => state.isHovering = false);

        btnFullscreen.addEventListener('click', async () => {
            try {
                if (!document.fullscreenElement && !document.webkitFullscreenElement) {
                    const el = document.documentElement;
                    const requestFS = el.requestFullscreen || el.webkitRequestFullscreen;
                    
                    if (requestFS) {
                        await requestFS.call(el);
                        iconFullscreen.setAttribute('data-lucide', 'minimize');
                        
                        if (screen.orientation && screen.orientation.lock) {
                            try { await screen.orientation.lock('landscape'); } 
                            catch (e) { console.warn("Orientación horizontal no soportada."); }
                        }
                    }
                } else {
                    const exitFS = document.exitFullscreen || document.webkitExitFullscreen;
                    if (exitFS) {
                        await exitFS.call(document);
                        iconFullscreen.setAttribute('data-lucide', 'maximize');
                        
                        if (screen.orientation && screen.orientation.unlock) {
                            screen.orientation.unlock();
                        }
                    }
                }
                lucide.createIcons();
            } catch (err) {
                console.warn("Pantalla completa no permitida:", err.message);
                if (screen.orientation && screen.orientation.lock) {
                    screen.orientation.lock('landscape').catch(e => console.warn(e));
                }
            }
        });

        function resizeCanvas() {
            const rtaRect = rtaCanvas.parentElement.getBoundingClientRect();
            const dpr = window.devicePixelRatio || 1;
            rtaCanvas.width = rtaRect.width * dpr;
            rtaCanvas.height = rtaRect.height * dpr;
            rtaCtx.scale(dpr, dpr);
            
            const wRect = waterfallCanvas.parentElement.getBoundingClientRect();
            waterfallCanvas.width = wRect.width * dpr;
            waterfallCanvas.height = wRect.height * dpr;
            
            if (state.isRunning) {
                waterfallCtx.fillStyle = '#050505';
                waterfallCtx.fillRect(0,0, waterfallCanvas.width, waterfallCanvas.height);
                
                const newWidth = Math.floor(rtaRect.width);
                if (!peakArray || peakArray.length !== newWidth) {
                    peakArray = new Float32Array(newWidth);
                    peakArray.fill(0);
                }
            }

            drawGrid(); 
            renderLabels(); 
        }

        window.addEventListener('resize', () => setTimeout(resizeCanvas, 100));

        function renderLabels() {
            freqLabelsContainer.innerHTML = '';
            const freqs = [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000];
            
            freqs.forEach(f => {
                const xPct = (Math.log(f / state.minFreq) / Math.log(state.maxFreq / state.minFreq)) * 100;
                
                if(xPct > 2 && xPct < 98) {
                    const span = document.createElement('span');
                    span.className = 'text-[7px] md:text-[8px] font-black text-slate-500 uppercase tracking-[0.1em] absolute transform -translate-x-1/2';
                    span.style.left = `${xPct}%`;
                    span.textContent = f >= 1000 ? `${f/1000}k` : f;
                    freqLabelsContainer.appendChild(span);
                }
            });
        }

        function drawGrid() {
            const width = rtaCanvas.width / (window.devicePixelRatio || 1);
            const height = rtaCanvas.height / (window.devicePixelRatio || 1);
            
            rtaCtx.fillStyle = '#050505';
            rtaCtx.fillRect(0, 0, width, height);

            rtaCtx.strokeStyle = 'rgba(255, 255, 255, 0.04)'; 
            rtaCtx.lineWidth = 1;
            
            const gridFreqs = [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000];
            rtaCtx.beginPath();
            gridFreqs.forEach(f => {
                const x = width * (Math.log(f / state.minFreq) / Math.log(state.maxFreq / state.minFreq));
                rtaCtx.moveTo(x, 0);
                rtaCtx.lineTo(x, height);
            });
            rtaCtx.stroke();

            rtaCtx.strokeStyle = 'rgba(255, 255, 255, 0.03)';
            rtaCtx.beginPath();
            for (let y = 0; y <= height; y += height / 6) {
                rtaCtx.moveTo(0, y);
                rtaCtx.lineTo(width, y);
            }
            rtaCtx.stroke();
        }

        function dspLoop() {
            if (!state.isRunning || state.isFrozen) {
                state.animationId = requestAnimationFrame(dspLoop);
                return;
            }

            if (!dataArray || !peakArray) {
                state.animationId = requestAnimationFrame(dspLoop);
                return;
            }

            const width = Math.floor(rtaCanvas.width / (window.devicePixelRatio || 1));
            const height = rtaCanvas.height / (window.devicePixelRatio || 1);
            const wPhysicalWidth = waterfallCanvas.width;
            const wPhysicalHeight = waterfallCanvas.height;

            analyser.getByteFrequencyData(dataArray);
            drawGrid();

            let frameMaxVal = -1;
            let frameMaxFreqHz = 0;
            const sampleRate = audioCtx.sampleRate;
            const bufferLength = analyser.frequencyBinCount;
            const fftSize = analyser.fftSize;
            const dbRange = analyser.maxDecibels - analyser.minDecibels;

            waterfallCtx.drawImage(waterfallCanvas, 0, 0, wPhysicalWidth, wPhysicalHeight - 1, 0, 1, wPhysicalWidth, wPhysicalHeight - 1);
            waterfallCtx.fillStyle = '#050505';
            waterfallCtx.fillRect(0, 0, wPhysicalWidth, 1);
            
            const newRow = waterfallCtx.createImageData(wPhysicalWidth, 1);
            const rowData = newRow.data;
            
            const barWidth = 1; 
            
            for (let x = 0; x < width; x += barWidth) {
                const freq = state.minFreq * Math.pow(state.maxFreq / state.minFreq, x / width);
                const binExacto = (freq * fftSize) / sampleRate;
                const b0 = Math.floor(binExacto);
                const b1 = Math.min(b0 + 1, bufferLength - 1);
                const fraccion = binExacto - b0;
                
                let valRaw = dataArray[b0] * (1 - fraccion) + dataArray[b1] * fraccion;
                const tiltCompensation = Math.max(0, 1 - Math.log10(freq / 20) / Math.log10(20000 / 20));
                let val = valRaw * (1 - tiltCompensation * 0.2);
                
                if (valRaw > frameMaxVal && freq > 40) { 
                    frameMaxVal = valRaw;
                    frameMaxFreqHz = freq;
                }

                const barHeight = Math.pow(val / 255, 1.4) * (height - 60);
                const hue = 240 - ((x / width) * 240);
                const alpha = val < 10 ? 0.2 : 0.8;
                
                rtaCtx.fillStyle = `hsla(${hue}, 100%, 60%, ${alpha})`;
                rtaCtx.fillRect(x, height - barHeight - 32, barWidth, barHeight);

                if (x < peakArray.length) {
                    if (val > peakArray[x]) {
                        peakArray[x] = val; 
                    } else {
                        peakArray[x] -= 1.5; 
                    }

                    const peakHeight = Math.pow(peakArray[x] / 255, 1.4) * (height - 60);
                    if(peakHeight > 2) {
                        rtaCtx.fillStyle = '#ffffff';
                        rtaCtx.globalAlpha = 0.8;
                        rtaCtx.fillRect(x, height - peakHeight - 34, barWidth, 1.5);
                        rtaCtx.globalAlpha = 1.0;
                    }
                }

                const ratio = wPhysicalWidth / width;
                const wX_start = Math.floor(x * ratio);
                const wX_end = Math.ceil((x + barWidth) * ratio);
                
                for(let wx = wX_start; wx < wX_end && wx < wPhysicalWidth; wx++) {
                    const idx = wx * 4;
                    if (valRaw < 5) {
                        rowData[idx] = 5; rowData[idx+1] = 5; rowData[idx+2] = 5; rowData[idx+3] = 255;
                    } else {
                        const wHue = (1 - (valRaw/255)) * 240; 
                        const s = 1;
                        const l = valRaw/255 * 0.6; 
                        
                        const c = (1 - Math.abs(2 * l - 1)) * s;
                        const x_color = c * (1 - Math.abs((wHue / 60) % 2 - 1));
                        const m = l - c/2;
                        let r, g, b;
                        
                        if (wHue < 60) { r=c; g=x_color; b=0;}
                        else if (wHue < 120) { r=x_color; g=c; b=0;}
                        else if (wHue < 180) { r=0; g=c; b=x_color;}
                        else if (wHue < 240) { r=0; g=x_color; b=c;}
                        else { r=x_color; g=0; b=c;}

                        rowData[idx] = Math.round((r+m)*255);
                        rowData[idx+1] = Math.round((g+m)*255);
                        rowData[idx+2] = Math.round((b+m)*255);
                        rowData[idx+3] = 255; 
                    }
                }
            }

            waterfallCtx.putImageData(newRow, 0, 0);

            // --- RENDERIZADO DEL CURSOR INTERACTIVO (CROSSHAIR) MEJORADO ---
            if (state.isHovering && state.hoverX >= 0 && state.hoverX <= width) {
                rtaCtx.globalAlpha = 1.0;
                
                const hoverFreq = state.minFreq * Math.pow(state.maxFreq / state.minFreq, state.hoverX / width);
                
                const binHover = (hoverFreq * fftSize) / sampleRate;
                const hb0 = Math.floor(binHover);
                const hb1 = Math.min(hb0 + 1, bufferLength - 1);
                const hFrac = binHover - hb0;
                const valHoverRaw = dataArray[hb0] * (1 - hFrac) + dataArray[hb1] * hFrac;
                
                const hoverDb = analyser.minDecibels + (valHoverRaw / 255) * dbRange;

                // Línea de escaneo láser (más visible)
                rtaCtx.beginPath();
                rtaCtx.strokeStyle = 'rgba(57, 255, 20, 0.8)'; 
                rtaCtx.lineWidth = 1;
                rtaCtx.moveTo(state.hoverX, 0);
                rtaCtx.lineTo(state.hoverX, height - 32); 
                rtaCtx.stroke();
                
                const hBarHeight = Math.pow(valHoverRaw / 255, 1.4) * (height - 60);
                const intersectionY = height - hBarHeight - 32;
                
                // Círculo de mira sobre la frecuencia
                rtaCtx.beginPath();
                rtaCtx.arc(state.hoverX, intersectionY, 4, 0, 2 * Math.PI);
                rtaCtx.fillStyle = '#ffffff';
                rtaCtx.fill();
                rtaCtx.shadowBlur = 12;
                rtaCtx.shadowColor = '#39FF14';
                rtaCtx.stroke();
                rtaCtx.shadowBlur = 0; 

                // Textos del Tooltip
                const labelFreq = hoverFreq >= 1000 ? (hoverFreq/1000).toFixed(2) + ' kHz' : Math.round(hoverFreq) + ' Hz';
                const labelDb = hoverDb <= analyser.minDecibels + 1 ? '--- dBFS' : hoverDb.toFixed(1) + ' dB';
                const tooltipText = `${labelFreq} | ${labelDb}`;
                
                // TIPOGRAFÍA MÁS GRANDE Y AUDAZ PARA MAYOR CLARIDAD (14px, 900 weight)
                rtaCtx.font = '900 14px "JetBrains Mono", monospace';
                const textMetrics = rtaCtx.measureText(tooltipText);
                const padding = 12;
                const boxWidth = textMetrics.width + (padding * 2);
                const boxHeight = 34;
                
                let boxX = state.hoverX + 15;
                if (boxX + boxWidth > width) {
                    boxX = state.hoverX - boxWidth - 15;
                }
                
                let boxY = intersectionY - 45;
                if (boxY < 10) boxY = 10; 
                
                // Fondo del Tooltip (más oscuro para mejor contraste)
                rtaCtx.fillStyle = 'rgba(3, 3, 3, 0.95)';
                rtaCtx.strokeStyle = 'rgba(57, 255, 20, 0.6)';
                rtaCtx.lineWidth = 1.5;
                rtaCtx.beginPath();
                rtaCtx.roundRect(boxX, boxY, boxWidth, boxHeight, 6);
                rtaCtx.fill();
                rtaCtx.stroke();
                
                // Texto con sutil efecto neón
                rtaCtx.fillStyle = '#39FF14';
                rtaCtx.shadowBlur = 8;
                rtaCtx.shadowColor = 'rgba(57, 255, 20, 0.4)';
                rtaCtx.fillText(tooltipText, boxX + padding, boxY + 22);
                rtaCtx.shadowBlur = 0; // Restaurar sombras
                rtaCtx.lineWidth = 1;  // Restaurar bordes
            }

            if (frameMaxVal > 20) {
                state.peakHz = Math.round(frameMaxFreqHz);
                valPeakFreq.textContent = state.peakHz.toString().padStart(4, '0');
                
                const dbfs = analyser.minDecibels + (frameMaxVal / 255) * dbRange;
                valPeakMag.textContent = dbfs.toFixed(1);
            } else {
                valPeakFreq.textContent = "----";
                valPeakMag.textContent = "--.-";
            }

            state.animationId = requestAnimationFrame(dspLoop);
        }

        async function initAudioEngine() {
            try {
                const el = document.documentElement;
                const requestFS = el.requestFullscreen || el.webkitRequestFullscreen;
                
                if (!document.fullscreenElement && requestFS) {
                    try {
                        await requestFS.call(el);
                        iconFullscreen.setAttribute('data-lucide', 'minimize');
                        lucide.createIcons();
                        if (screen.orientation && screen.orientation.lock) {
                            await screen.orientation.lock('landscape');
                        }
                    } catch(e) { 
                        console.warn("Auto-pantalla completa omitida por el entorno."); 
                    }
                }

                const stream = await navigator.mediaDevices.getUserMedia({
                    audio: { echoCancellation: false, noiseSuppression: false, autoGainControl: false }
                });

                startOverlay.style.opacity = '0';
                setTimeout(() => startOverlay.style.display = 'none', 500);

                const AudioContext = window.AudioContext || window.webkitAudioContext;
                audioCtx = new AudioContext();
                state.maxFreq = audioCtx.sampleRate / 2; 

                analyser = audioCtx.createAnalyser();
                analyser.fftSize = 4096; 
                analyser.smoothingTimeConstant = 0.5; 
                analyser.minDecibels = -100;
                analyser.maxDecibels = 0;

                const highpass = audioCtx.createBiquadFilter();
                highpass.type = 'highpass';
                highpass.frequency.value = 20;
                highpass.Q.value = 0.7;

                const source = audioCtx.createMediaStreamSource(stream);
                source.connect(highpass);
                highpass.connect(analyser);

                const bufferLength = analyser.frequencyBinCount;
                dataArray = new Uint8Array(bufferLength);
                
                state.isRunning = true;
                resizeCanvas();
                dspLoop();

            } catch (err) {
                console.error("Error inicializando motor DSP:", err);
                alert("Acceso denegado al micrófono o error de hardware.");
            }
        }

        btnStart.addEventListener('click', () => {
            if (!state.isRunning) initAudioEngine();
        });
        
        startOverlay.addEventListener('click', (e) => {
             if (e.target.id !== 'btn-start' && !state.isRunning) initAudioEngine();
        });

        btnFreeze.addEventListener('click', () => {
            if (!state.isRunning) return;
            state.isFrozen = !state.isFrozen;
            
            if (state.isFrozen) {
                freezeGlow.style.opacity = '0.2';
                btnFreeze.classList.add('border-[#39FF14]');
                iconFreeze.setAttribute('data-lucide', 'play');
                iconFreeze.classList.add('text-[#39FF14]');
            } else {
                freezeGlow.style.opacity = '0';
                btnFreeze.classList.remove('border-[#39FF14]');
                iconFreeze.setAttribute('data-lucide', 'pause');
                iconFreeze.classList.remove('text-[#39FF14]');
            }
            lucide.createIcons(); 
        });

        resizeCanvas();
    </script>
